public sealed class ExperimentGate
{
    private readonly SemaphoreSlim _busy = new(1, 1);

    public async Task<T> Run<T>(Func<Task<T>> experiment)
    {
        if (!_busy.Wait(0)) throw new InvalidOperationException("An experiment is already running. Wait for it to finish.");
        try { return await experiment(); }
        finally { _busy.Release(); }

3 · Apply the Multi-turn Sessions and Message Queues rules

The local reducer rejects blank messages, reports a conflict when a competing writer is present, and otherwise issues one receipt per message. Its idle/end_turn status is authored data, not an observation.

Source: src/Lesson.C10/Lessons/C10Lesson.cs · starting at line 26

public Task<Result> Run(JsonObject input)
{
    var messages = List(input, "messages");
    if (messages.Any(m => m is not JsonValue v || !v.TryGetValue(out string? s) || string.IsNullOrWhiteSpace(s)))
        throw new ArgumentException("Each queued message must contain text");

    if (Flag(input, "concurrentInput"))
        return Task.FromResult(Report(Outcome.Conflict, "Another writer invalidated exclusive turn attribution",
            new JsonObject { ["turns"] = 0 },
            "Admit one writer", "Observe a foreign user event", "Block queue advancement"));

    var receipts = new JsonArray(messages.Select((message, index) => (JsonNode)new JsonObject
    {
        ["commandId"] = $"cmd-{index + 1}",
        ["userEventId"] = $"user-{index + 1}",
        ["message"] = message!.GetValue<string>(),
        ["status"] = "idle/end_turn",
    }).ToArray());

    return Task.FromResult(Report(Outcome.Passed, "Serialized each admitted turn in one conversation",
        new JsonObject { ["turns"] = receipts.Count, ["receipts"] = receipts },
        "Open one shared stream reader", "Send exactly one queued message", "Match the accepted user event ID", "Confirm reusable idle before advancing"));

4.1 · Live queue: open the stream before sending

This is the step the local experiment only describes. The live queue drives a real Managed Agents session; API use may incur charges.

SSE has no replay. The C# stream is a lazy IAsyncEnumerable, so a background task starts it and buffers every event in a channel before the first message is sent.

Source: src/Lesson.C10/Services/SessionStream.cs · starting at line 72

public SessionStream(AnthropicClient client, string sessionId, TimeSpan timeout, CancellationToken cancellationToken)
{
    _stop = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
    var streamClient = client.WithOptions(options => options with { Timeout = timeout });
    _reader = Task.Run(async () =>
    {
        try
        {
            await foreach (var item in streamClient.Beta.Sessions.Events.StreamStreaming(sessionId, new EventStreamParams(), _stop.Token))
                await _events.Writer.WriteAsync(item, _stop.Token);
            _events.Writer.TryComplete();
        }
        catch (Exception error) { _events.Writer.TryComplete(error); }
    });

4.2 · Live queue: send exactly one message and correlate it

Each send returns the server-assigned user event ID, which becomes the command's correlation key. With Concurrent Input checked, a second writer posts into the same session and the queue never learns that ID. The queue reads events only after the send returns, so its own ID is always known first.

Source: src/Lesson.C10/Services/LiveQueue.cs · starting at line 49

for (var index = 0; index < messages.Count; index++)
{
    // Exactly one queued message per turn; the server-assigned event ID is the correlation key.
    var commandId = $"cmd-{index + 1}";
    var userEventId = await SendText(client, session.ID, messages[index], deadline.Token);
    ownEventIds.Add(userEventId);
    await Teach(Teaching.Sent(index + 1, commandId, messages[index], userEventId));
    if (competingWriter && index == 0)
    {
        // A second writer posts into the same conversation. The queue never learns this ID.
        await SendText(client, session.ID, SecondWriterMessage, deadline.Token);
        await Teach(Teaching.SecondWriter(SecondWriterMessage));
    }

    var turn = new TurnTracker(userEventId, ownEventIds);
    var thinkingShown = false;
    await foreach (var item in events.ReadAll(deadline.Token))
    {
        eventTypes.Add(item.Type);
        var state = turn.Observe(SessionStream.ToSignal(item));
        // Thinking can arrive as several events per turn; one explanation is enough.
        if (turn.Last != Observation.Thinking || !thinkingShown)
            await Teach(Teaching.Describe(turn.Last, turn, index + 1, commandId, index + 1 < messages.Count, turn.ForeignUserEventId));
        thinkingShown |= turn.Last == Observation.Thinking;
        if (state != TurnState.Waiting || eventTypes.Count > 500) break;
    }

    receipts.Add(new QueueReceipt(commandId, userEventId, messages[index], turn.QueuedSeen, turn.Processed, turn.State switch
    {
        TurnState.Settled => "idle/end_turn",
        // Claude may have answered, but the turn never got an end_turn idle of its own.
        TurnState.Conflict => turn.Answered ? "answered, not settled: foreign user event" : "blocked: foreign user event",
        TurnState.Stopped => $"stopped: {turn.StopReason}",
        _ => "unsettled",
    }, turn.Answer));

4.3 · Live queue: advance only after a settled idle

TurnTracker has no SDK types, so it is unit-tested against recorded event sequences. Classify decides what each event means for the current turn: a foreign user event is a conflict; an idle seen before our message was processed belongs to an earlier turn; requires_action is a transient idle; only end_turn settles the turn.

Source: src/Lesson.C10/Services/TurnTracker.cs · starting at line 46

Observation Classify(SessionSignal signal) => signal.Type switch
{
    "user.message" when signal.Id == UserEventId => signal.Processed ? Observation.OwnProcessed : Observation.OwnQueued,
    "user.message" when signal.Id is not null && ownEventIds.Contains(signal.Id) => Observation.EarlierOwnEvent,
    "user.message" when signal.Id is not null => Observation.Foreign,
    "agent.thinking" when Processed => Observation.Thinking,
    "agent.message" when Processed => Observation.Answer,
    // An idle seen before our message was processed belongs to an earlier turn.
    "session.status_idle" when !Processed => Observation.StaleIdle,
    // requires_action is a transient idle (waiting on the client), never a finished turn.
    "session.status_idle" when signal.StopReason == "requires_action" => Observation.TransientIdle,
    "session.status_idle" when signal.StopReason == "end_turn" => Observation.Settled,
    "session.status_idle" => Observation.Stopped,
    "session.status_terminated" => Observation.Stopped,
    _ => Observation.Ignored,

public static TeachingStep? Describe(Observation observation, TurnTracker turn, int turnNumber, string commandId, bool hasNext, string? foreignId) => observation switch
{
    Observation.OwnQueued => new(Concept.Correlate, Tone.Wait, $"Event {turn.UserEventId} is queued",
        "The session has received our message but is still busy with earlier work (processed_at is empty)."),
    Observation.OwnProcessed => new(Concept.Correlate, Tone.Good, $"The session picked up {commandId}'s event",
        $"Event {turn.UserEventId} matches the receipt for {commandId}, so what Claude says next belongs to this command."),
    Observation.Foreign => new(Concept.OneWriter, Tone.Stop, $"Unknown user event {foreignId}: another writer!",

    Observation.Thinking => new(Concept.Settled, Tone.Wait, "Claude is thinking",

    Observation.Answer => new(Concept.Settled, Tone.Info, "Claude answered",

What the evidence proves

The local experiment is an in-process serialized queue with authored turn events. It demonstrates admission and attribution, not a durable distributed queue. The run journal stores results, not production workflow state.

The live queue observes a real session: accepted user event IDs, processed echoes, each turn's answer, and the idle/end_turn that allowed the next message. It still has a single in-process writer, so it does not show a durable or distributed queue. Check the outcome, the receipts and the answers together. A rendered answer alone does not prove that a tool ran, a remote operation succeeded, or work was cancelled.
